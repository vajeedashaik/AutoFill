\section*{Selected Projects}

\textbf{Urban Traffic Congestion Prediction -- ML Pipeline \& Dashboard}
\hfill \textit{IEEE ICAFT 2025}

\begin{itemize}
    \item Cleaned 48K hourly sensor records with \textbf{pandas} (deduplication, median imputation of impossible readings, outlier clipping) and engineered \textbf{27 features}: lag and rolling volume, cyclical time encodings, weather, calendar.
    \item Compared HistGradientBoosting, Random Forest, and ExtraTrees in \textbf{scikit-learn}; the best model reached \textbf{93.7\% accuracy (0.937 weighted F1)} on a held-out 15\% stratified split, with permutation importance showing lag and time-of-day features carry about 76\% of the signal.
    \item Wrapped inference (1--3 ms) in a 4-stage pipeline exposed through 4 \textbf{FastMCP} servers (22 tools); delivered an 11-page \textbf{Streamlit} dashboard with \textbf{Docker} and \textbf{Supabase} data loading that falls back to CSV.
\end{itemize}

\textbf{BovineID -- Cattle Breed Classifier (41 Indian Breeds)}
\hfill \textit{Deployed ML web app}

\begin{itemize}
    \item Trained a \textbf{VGG16 transfer-learning} classifier in \textbf{TensorFlow/Keras} (frozen base, dense head, early stopping, learning-rate scheduling): \textbf{78.4\% top-1 and 92.2\% top-5 accuracy, 0.749 macro F1} on a stratified 20\% validation split (1,165 images).
